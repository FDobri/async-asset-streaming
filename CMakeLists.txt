cmake_minimum_required(VERSION 3.24)
project(asset_stream LANGUAGES C CXX)
include(FetchContent)
include(CTest)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(STREAM_DEMO "Build the SDL3/OpenGL streaming scene" ON)
option(STREAM_TRACY "Enable Tracy zones" OFF)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
FetchContent_Declare(zstd URL https://codeload.github.com/facebook/zstd/tar.gz/refs/tags/v1.5.7 SOURCE_SUBDIR build/cmake)
FetchContent_MakeAvailable(zstd)
add_library(stream src/stream.cpp)
target_include_directories(stream PUBLIC include)
target_link_libraries(stream PRIVATE libzstd_static Threads::Threads)
if(STREAM_TRACY)
    FetchContent_Declare(tracy URL https://codeload.github.com/wolfpld/tracy/tar.gz/refs/tags/v0.11.1)
    FetchContent_MakeAvailable(tracy)
    target_link_libraries(stream PRIVATE Tracy::TracyClient)
    target_compile_definitions(stream PRIVATE STREAM_TRACY)
endif()
add_executable(asset_pack pack.cpp)
target_link_libraries(asset_pack PRIVATE stream)
if(BUILD_TESTING)
    add_executable(stream_test tests.cpp)
    target_link_libraries(stream_test PRIVATE stream)
    add_test(NAME stream_test COMMAND stream_test)
endif()
if(STREAM_DEMO)
    find_package(OpenGL REQUIRED)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL3 URL https://codeload.github.com/libsdl-org/SDL/tar.gz/refs/tags/release-3.2.10)
    FetchContent_Declare(imgui URL https://codeload.github.com/ocornut/imgui/tar.gz/refs/tags/v1.91.9b)
    FetchContent_MakeAvailable(SDL3 imgui)
    add_executable(stream_scene demo.cpp
        ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
        ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
        ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp
        ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl2.cpp)
    target_include_directories(stream_scene PRIVATE ${imgui_SOURCE_DIR})
    target_link_libraries(stream_scene PRIVATE stream SDL3::SDL3 OpenGL::GL)
endif()
